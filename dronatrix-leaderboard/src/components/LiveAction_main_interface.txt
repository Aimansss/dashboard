  // MAIN INTERFACE
  return (
    <div className="h-full flex flex-col p-4 bg-dark-950">
      {/* Team Header - Compact */}
      <motion.div
        initial={{ y: -20, opacity: 0 }}
        animate={{ y: 0, opacity: 1 }}
        className="text-center mb-3"
      >
        <div className="flex items-center justify-center gap-3 mb-1">
          <div className="px-2 py-1 bg-dark-800 border-l-2 border-primary">
            <span className="text-xs font-bold text-gray-500 uppercase tracking-wider">R{currentRound}</span>
          </div>
          <h1 className="text-3xl font-black text-primary tracking-tight uppercase" style={{letterSpacing: '-0.02em'}}>
            {currentTeam.name}
          </h1>
        </div>
        <div className="text-gray-600 text-xs font-mono">
          {currentTeam.members.join(' • ')}
        </div>
      </motion.div>

      {/* Main Content */}
      <div className="flex-1 min-h-0 flex flex-col gap-2">
        {isDeclarationRound ? (
          // Declaration Rounds (R2/R4) - Two Column Layout
          <div className="grid grid-cols-2 gap-3 flex-1">
            {/* Left Column */}
            <div className="flex flex-col gap-2">
              {/* Auto-selected Pilot */}
              {availablePilots.length === 1 && (
                <div className="bg-dark-800/50 border-l-2 border-accent-success p-2">
                  <div className="text-xs font-bold text-gray-600 uppercase tracking-wider">Pilot</div>
                  <div className="text-sm font-bold text-gray-100">{pilot}</div>
                </div>
              )}

              {/* Mode Selection */}
              <div className="bg-dark-800/50 border-l-2 border-primary/50 p-2 flex-1">
                <label className="block text-xs font-bold text-gray-600 mb-2 uppercase tracking-wider">Mode</label>
                <div className="grid grid-cols-1 gap-2">
                  {Object.entries(roundModes).map(([key, config]) => (
                    <button
                      key={key}
                      onClick={() => setMode(key)}
                      className={`p-2 border transition-all text-left ${
                        mode === key
                          ? 'border-primary bg-primary/10'
                          : 'border-dark-700 bg-dark-900 hover:border-primary/30'
                      }`}
                    >
                      <div className={`font-black text-sm uppercase ${config.color}`}>
                        {config.label}
                      </div>
                      <div className="text-xs font-mono text-gray-600">
                        +{config.successPoints} / {config.failPoints}
                      </div>
                    </button>
                  ))}
                </div>
              </div>
            </div>

            {/* Right Column */}
            <div className="flex flex-col gap-2">
              {/* Target Display */}
              {mode && (
                <div className="bg-dark-800/50 border-l-2 border-accent-success p-2">
                  <div className="text-xs font-bold text-gray-600 mb-1 uppercase tracking-wider">Challenge</div>
                  <div className="text-sm font-bold text-gray-100 mb-1">
                    {getTargetDisplay()}
                  </div>
                  <div className="text-xs text-gray-500">
                    {currentRound === 2 && mode === 'SAFE' && `Your R1: ${formatTime(currentTeam.rounds[1].time)}`}
                    {currentRound === 2 && mode === 'RISKY' && `Your R1: ${formatTime(currentTeam.rounds[1].time)} — Beat by 10s`}
                    {currentRound === 2 && mode === 'ALL_IN' && `Fastest R1: ${formatTime(getFastestRound1())}`}
                    {currentRound === 4 && mode === 'SAFE' && `Your R3: ${formatTime(currentTeam.rounds[3].time)} — Beat by 5s`}
                    {currentRound === 4 && mode === 'CHALLENGE' && rival && (() => {
                      const rivalTeam = teams.find(t => t.id === parseInt(rival));
                      return `${rivalTeam?.name} R3: ${formatTime(rivalTeam?.rounds[3].time)}`;
                    })()}
                    {currentRound === 4 && mode === 'BLOOD_MATCH' && (() => {
                      const fastest = getFastestRound3Team();
                      return `${fastest?.name} R3: ${formatTime(fastest?.rounds[3].time)}`;
                    })()}
                  </div>
                </div>
              )}

              {/* Rival Selection */}
              {(mode === 'CHALLENGE' || mode === 'BLOOD_MATCH') && (
                <div className="bg-dark-800/50 border-l-2 border-primary/50 p-2">
                  <label className="block text-xs font-bold text-gray-600 mb-1 uppercase tracking-wider">
                    {mode === 'BLOOD_MATCH' ? 'Target' : 'Rival'}
                  </label>
                  <select
                    value={rival}
                    onChange={(e) => setRival(e.target.value)}
                    className="w-full bg-dark-900 border border-dark-700 px-3 py-2 text-gray-100 text-sm font-bold focus:outline-none focus:border-primary disabled:opacity-50"
                    disabled={mode === 'BLOOD_MATCH'}
                  >
                    <option value="">Select Rival</option>
                    {teams
                      .filter(t => t.id !== currentTeam.id && t.rounds[3].time !== null)
                      .map((t) => (
                        <option key={t.id} value={t.id}>
                          {t.name} — {formatTime(t.rounds[3].time)}
                        </option>
                      ))}
                  </select>
                  {mode === 'BLOOD_MATCH' && (() => {
                    const fastest = getFastestRound3Team();
                    if (fastest) {
                      setTimeout(() => setRival(fastest.id.toString()), 0);
                    }
                  })()}
                </div>
              )}

              {/* Timer */}
              <div className="bg-dark-800/50 border-l-2 border-primary p-2 flex-1 flex flex-col">
                <div className="flex-1 flex items-center justify-center">
                  <div className="text-4xl font-black font-mono text-primary text-center" style={{letterSpacing: '-0.05em'}}>
                    {formatTime(elapsedTime)}
                  </div>
                </div>
                <div className="flex gap-2">
                  <button
                    onClick={startTimer}
                    disabled={timerRunning}
                    className="flex-1 px-4 py-2 bg-accent-success text-dark-950 text-sm font-black uppercase tracking-wide hover:bg-accent-success/90 transition-all disabled:opacity-30"
                  >
                    START
                  </button>
                  <button
                    onClick={stopTimer}
                    disabled={!timerRunning}
                    className="flex-1 px-4 py-2 bg-accent-danger text-white text-sm font-black uppercase tracking-wide hover:bg-accent-danger/90 transition-all disabled:opacity-30"
                  >
                    STOP
                  </button>
                </div>
              </div>
            </div>
          </div>
        ) : (
          // Time Trial Rounds (R1/R3) - Single Column
          <div className="flex flex-col gap-2">
            {/* Pilot Selection */}
            <div className="bg-dark-800/50 border-l-2 border-primary/50 p-3">
              <label className="block text-xs font-bold text-gray-600 mb-2 uppercase tracking-wider">Select Pilot</label>
              <div className="flex gap-2">
                {availablePilots.map((p) => (
                  <button
                    key={p}
                    onClick={() => setPilot(p)}
                    className={`flex-1 px-4 py-2 text-sm font-bold uppercase tracking-wide transition-all ${
                      pilot === p
                        ? 'bg-primary text-dark-950'
                        : 'bg-dark-900 text-gray-500 hover:bg-dark-700 hover:text-gray-300'
                    }`}
                  >
                    {p}
                  </button>
                ))}
              </div>
            </div>

            {/* Time Input */}
            <div className="bg-dark-800/50 border-l-2 border-primary p-4 flex-1 flex flex-col justify-center">
              <label className="block text-xs font-bold text-gray-600 mb-3 uppercase tracking-wider text-center">Enter Time</label>
              <div className="flex items-center justify-center gap-2 mb-2">
                <input
                  type="number"
                  min="0"
                  max="9"
                  value={minutes}
                  onChange={(e) => setMinutes(e.target.value.slice(0, 1))}
                  placeholder="0"
                  className="w-20 bg-dark-900 border border-dark-700 px-4 py-3 text-3xl font-black font-mono text-primary text-center focus:outline-none focus:border-primary"
                />
                <span className="text-2xl font-bold text-gray-600">:</span>
                <input
                  type="number"
                  min="0"
                  max="59"
                  value={seconds}
                  onChange={(e) => setSeconds(e.target.value.slice(0, 2))}
                  placeholder="00"
                  className="w-24 bg-dark-900 border border-dark-700 px-4 py-3 text-3xl font-black font-mono text-primary text-center focus:outline-none focus:border-primary"
                />
                <span className="text-2xl font-bold text-gray-600">.</span>
                <input
                  type="number"
                  min="0"
                  max="99"
                  value={milliseconds}
                  onChange={(e) => setMilliseconds(e.target.value.slice(0, 2))}
                  placeholder="00"
                  className="w-24 bg-dark-900 border border-dark-700 px-4 py-3 text-3xl font-black font-mono text-primary text-center focus:outline-none focus:border-primary"
                />
              </div>
              <div className="text-center text-xs text-gray-600 uppercase tracking-wider">
                MIN : SEC . CENTISEC
              </div>
            </div>
          </div>
        )}

        {/* Error Display */}
        {error && (
          <div className="p-2 bg-accent-danger/10 border-l-2 border-accent-danger text-accent-danger text-xs font-bold">
            {error}
          </div>
        )}
      </div>

      {/* Fixed Submit Button at Bottom */}
      <div className="border-t border-dark-700 pt-3">
        <button
          onClick={handleSubmit}
          className="w-full px-8 py-3 bg-primary text-dark-950 text-base font-black uppercase tracking-wide hover:bg-primary-light transition-all"
        >
          {isDeclarationRound ? 'SUBMIT' : 'SUBMIT & NEXT TEAM'}
        </button>
      </div>
    </div>
  );
